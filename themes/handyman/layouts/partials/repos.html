Live data comes from the public GitHub API at build time; if that request fails we fall back to the curated list in hugo.toml so the build never breaks. Forks, archived repositories, the site repo itself, and (by default) repos with no description are filtered out. Control it via: params.github, params.reposLimit, params.repoExclude, params.repoPinned */}} {{- $site := . -}} {{- $user := $site.Site.Params.github -}} {{- $limit := $site.Site.Params.reposLimit | "Jupyter Notebook" "#DA5B0B" "Java" "#b07219" "Lua" "#000080" "Ruby" "#701516" "Nix" "#7e7eff" -}} {{- $headers := dict "Accept" "application/vnd.github+json" "User-Agent" (printf "%s-site" $user) -}} {{- warnf "repos.html: GitHub API request failed, using the fallback list: %s" .Err -}} {{- else -}} {{- range (.Value.Content | transform.Unmarshal) -}} {{- if and (not .fork) (not .archived) (not (strings.HasSuffix .name ".github.io")) (not (in $exclude .name)) .description -}} {{- $items = $items | append (dict "name" .name "url" .html_url "description" (.description | plainify) "language" $limit) (not (in $seen .name)) -}} {{- $featured = $featured | append . -}} {{- $seen = $seen | append .name -}} {{- end -}} {{- end -}} {{/* Last-resort fallback when the API returned nothing. */}} {{- if not $featured -}} {{- range $site.Site.Params.repos -}} {{- if lt (len $featured) $limit -}} {{- $featured = $featured | append (dict "name" .name "url" .url "description" .desc "language" (.language | default "") "pushed" "" "stars" 0) -}} {{- end -}} {{- end -}} {{- end -}} {{- if $featured -}}
